        }
    }, []);

    const handleLogin = (user: string) => {
        setIsLoggedIn(true);
        setUsername(user);
        localStorage.setItem('hackingPracticeUsername', user); // Store username (simplified)
        setCurrentPage('sql'); // Redirect to SQL Injection page after login
    };

    const handleLogout = () => {
        setIsLoggedIn(false);
        setUsername('');
        localStorage.removeItem('hackingPracticeUsername'); // Remove stored username
        setCurrentPage('login');
    };

    if (!isLoggedIn) {
        return <LoginPage onLogin={handleLogin} />;
    }

    return (
        <div className="bg-gray-900 min-h-screen">
            <header className="bg-black/10 backdrop-blur-md py-4 px-6 border-b border-white/10 flex justify-between items-center">
                <h1 className="text-2xl font-bold text-white">Hacking Practice</h1>
                <div className="flex items-center gap-4">
                    <span className="text-gray-300">Logged in as: <span className="font-semibold text-purple-400">{username}</span></span>
                    <Button
                        onClick={handleLogout}
                        className="bg-red-500/20 hover:bg-red-500/30 text-red-400"
                    >
                        Logout
                    </Button>
                </div>
            </header>

            <Tabs defaultValue="sql" value={currentPage} onValueChange={setCurrentPage} className="p-4">
                <TabsList className="mb-8 bg-black/5 backdrop-blur-md border border-white/10">
                    <TabsTrigger value="sql" className="text-white data-[state=active]:bg-purple-500/20 data-[state=active]:text-purple-300">
                        <FileText className="mr-2 h-4 w-4" /> SQL Injection
                    </TabsTrigger>
                    <TabsTrigger value="xss" className="text-white data-[state=active]:bg-purple-500/20 data-[state=active]:text-purple-300">
                        <FileText className="mr-2 h-4 w-4" /> XSS
                    </TabsTrigger>
                    <TabsTrigger value="upload" className="text-white data-[state=active]:bg-purple-500/20 data-[state=active]:text-purple-300">
                        <FileText className="mr-2 h-4 w-4" /> File Upload
                    </TabsTrigger>
                    <TabsTrigger value="command" className="text-white data-[state=active]:bg-purple-500/20 data-[state=active]:text-purple-300">
                        <FileText className="mr-2 h-4 w-4" /> Command Injection
                    </TabsTrigger>
                </TabsList>
                <TabsContent value="sql">
                    <SQLInjectionPage />
                </TabsContent>
                <TabsContent value="xss">
                    <XSSPage />
                </TabsContent>
                <TabsContent value="upload">
                    <FileUploadPage />
                </TabsContent>
                <TabsContent value="command">
                    <CommandInjectionPage />
                </TabsContent>
            </Tabs>
        </div>
    );
};

export default HackingPracticeApp;
